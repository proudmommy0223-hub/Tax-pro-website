        function calculateRefund() {
            const status = document.getElementById('calc-status').value;
            const income = parseFloat(document.getElementById('calc-income').value) || 0;
            const withheld = parseFloat(document.getElementById('calc-withheld').value) || 0;
            const dependents = parseInt(document.getElementById('calc-dependents').value) || 0;
            const isBusiness = document.getElementById('calc-business').value;

            let baseRefund = withheld;
            let childCredit = dependents * 2000;
            
            let estLower = baseRefund + childCredit;
            if (income > 0 && income < 35000 && dependents > 0) {
                estLower += 1500; // Earned Income Credit estimate
            }
            if (isBusiness === 'yes') {
                estLower = Math.max(0, estLower - 500);
            }

            let estUpper = estLower * 1.25 + 300;

            const formatMoney = (amount) => {
                return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(amount);
            };

            document.getElementById('calc-amount').innerText = `${formatMoney(estLower)} - ${formatMoney(estUpper)}`;
            document.getElementById('calc-result').classList.remove('hidden');
        }

        function applyEstimateToForm() {
            const amountText = document.getElementById('calc-amount').innerText;
            const notesField = document.getElementById('notes-field');
            notesField.value = `Attached Estimate from Calculator: ${amountText}`;
        }

        function submitIntakeForm(event) {
            event.preventDefault();
            document.getElementById('tax-intake-form').classList.add('hidden');
            document.getElementById('form-success').classList.remove('hidden');
        }
